const network: any[] = [];
const originalFetch = window.fetch.bind(window);
window.fetch = async (...args) => {
  const url = new URL(typeof args[0] === 'string' ? args[0] : args[0] instanceof URL ? args[0].href : args[0].url, location.href);
  const entry: any = { url: url.origin + url.pathname, method: args[1]?.method || 'GET' };
  network.push(entry);
  try { const response = await originalFetch(...args); entry.status = response.status; show(); return response; }
  catch(e: any) { entry.error = String(e.message); show(); throw e; }
};
const mode = new URLSearchParams(location.search).get('mode') === 'CLOUD_API' ? 'CLOUD_API' : 'GATEWAY';
const app = cloudbase.init({ env, region: 'ap-shanghai', endPointMode: mode, timeout: 12000, auth: { detectSessionInUrl: false } });

const results: any[] = [];
const output = document.querySelector('pre')!;
function show() { output.textContent = JSON.stringify({ origin: location.origin, mode, results, network }, null, 2); }
function summary(value: any) { return { code: value?.code ?? value?.error?.code, status: value?.status ?? value?.statusCode ?? value?.result?.status, requestId: value?.requestId, hasResult: value?.result !== undefined }; }
async function check(name: string, fn: () => Promise<any>, expect: 'deny' | 'disabled') {
  try {
    const value = await fn();
    const s = summary(value);
    const passed = expect === 'disabled' ? s.status === 503 : /PERMISSION|DENIED|AUTH|CREDENTIAL|FORBIDDEN/i.test(String(s.code || ''));
    results.push({ name, passed, returned: s }); show();
    if (!passed) throw new Error('Unexpected result; stop');
  } catch(e: any) {
    if (e.message === 'Unexpected result; stop') throw e;
    const passed = expect === 'deny' && /PERMISSION|DENIED|AUTH|CREDENTIAL|FORBIDDEN/i.test(String(e.code || '') + ' ' + String(e.message || ''));
    results.push({ name, passed, rejected: summary(e), message: String(e.message || '').slice(0,220) }); show();
    if (!passed) throw new Error('Boundary not demonstrated; stop');
  }
}
